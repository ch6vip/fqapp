cmake_minimum_required(VERSION 3.16)
project(fqapp_linux_bridge LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(SPEECHD REQUIRED IMPORTED_TARGET speech-dispatcher)
find_package(Threads REQUIRED)
add_library(fqapp_linux_bridge STATIC bridge.cc tts.cc)
set_target_properties(fqapp_linux_bridge PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(fqapp_linux_bridge PUBLIC cxx_std_17)
target_include_directories(fqapp_linux_bridge PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(fqapp_linux_bridge PUBLIC PkgConfig::GTK PkgConfig::SPEECHD Threads::Threads)
if(TARGET flutter)
 target_link_libraries(fqapp_linux_bridge PUBLIC flutter)
else()
 set(FLUTTER_LINUX_DIR "" CACHE PATH "SDK engine/linux-x64 directory containing flutter_linux/")
 target_include_directories(fqapp_linux_bridge PUBLIC ${FLUTTER_LINUX_DIR})
 target_link_libraries(fqapp_linux_bridge PUBLIC ${FLUTTER_LINUX_DIR}/libflutter_linux_gtk.so)
endif()
